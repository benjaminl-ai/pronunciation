let mediaRecorder; let audioChunks = []; let isRecording = false; let audioStream = null; // 儲存串流 async function toggleRecording() { const btn = document.getElementById('recordBtn'); const status = document.getElementById('status'); if (!isRecording) { audioStream = await navigator.mediaDevices.getUserMedia({ audio: true }); mediaRecorder = new MediaRecorder(audioStream); audioChunks = []; mediaRecorder.ondataavailable = e => audioChunks.push(e.data); mediaRecorder.onstop = analyzeSpeech; mediaRecorder.start(); isRecording = true; btn.innerText = "🛑 停止錄音"; btn.classList.add('recording'); status.innerText = "正在錄音中..."; } else { mediaRecorder.stop(); // 【重要：關閉麥克風】 if (audioStream) { audioStream.getTracks().forEach(track => track.stop()); audioStream = null; } isRecording = false; btn.innerText = "🎤 開始試讀"; btn.classList.remove('recording'); status.innerText = "AI 正在思考中..."; } } async function analyzeSpeech() { const word = document.getElementById('wordInput').value; const audioBlob = new Blob(audioChunks, { type: 'audio/webm' }); const reader = new FileReader(); reader.readAsDataURL(audioBlob); reader.onloadend = async () => { const base64Audio = reader.result.split(',')[1]; const payload = { contents: [{ parts: [ { text: `你是一位專業英文老師。請分析學生讀「${word}」的發音。` }, { inline_data: { mime_type: "audio/webm", data: base64Audio } } ] }] }; try { // 如果 v1 唔得，呢度可以換返 v1beta 試試 const url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=" + MY_SECRET_KEY; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const data = await response.json(); if (data.error) throw new Error(data.error.message); const feedbackText = data.candidates[0].content.parts[0].text; document.getElementById('feedback').innerHTML = feedbackText.replace(/\n/g, '
'); document.getElementById('feedback').style.display = "block"; document.getElementById('status').innerText = "分析完畢！"; } catch (error) { console.error(error); alert("錯誤: " + error.message); } }; }
